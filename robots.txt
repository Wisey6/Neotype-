# neotype.au
User-agent: *
Allow: /

# The owner dashboard is not for search engines. This is a courtesy signal to
# well-behaved crawlers, not a security control — /admin is protected by a
# password, which is what actually keeps it private.
Disallow: /admin
Disallow: /admin.html
Disallow: /api/
Disallow: /success
Disallow: /success.html

# Internal documents and test suites. Pages publishes the whole repository, so
# these paths exist; _redirects 301s them to the homepage, and this stops a
# crawler asking in the first place. Adding a new internal file? Add it in both.
Disallow: /README.md
Disallow: /CLOUDFLARE.md
Disallow: /PRICING.md
Disallow: /SETUP.md
Disallow: /MAIL-DNS.md
Disallow: /package.json
Disallow: /brainstorms/
Disallow: /.tests/

Sitemap: https://www.neotype.au/sitemap.xml
